# imgready — https://imgready.app
#
# Everything here is a free, public, client-side tool. We want it crawled,
# indexed and cited. There is nothing to gate.
#
# /beta/ is deliberately NOT disallowed: it carries <meta name="robots"
# content="noindex,nofollow"> plus a canonical to /. Blocking it here would
# stop crawlers reading that noindex, which is the opposite of what we want.

User-agent: *
Allow: /
# /r/{id}/{filename} exists only inside the service worker, which serves a
# processed image under a proper filename. There is never a page there, and
# a crawler without the service worker gets a 404. Google found the "/r/"
# prefix as a string in the JavaScript and tried it; this stops the retries.
Disallow: /r/
